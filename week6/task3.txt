
import java.util.*;

public class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int p = sc.nextInt();

        while (p-- > 0) {
            String s1 = sc.next();
            String s2 = sc.next();

            boolean found = false;

            for (char c : s1.toCharArray()) {
                if (s2.indexOf(c) != -1) {
                    found = true;
                    break;
                }
            }

            System.out.println(found ? "YES" : "NO");
        }

        sc.close();
    }
}
