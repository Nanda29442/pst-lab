
import java.util.*;

class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        List<Integer> result = new ArrayList<>();

        int n = s.length();
        int m = p.length();

        if (n < m) {
            return result;
        }

        int[] count = new int[26];

        for (int i = 0; i < m; i++) {
            count[p.charAt(i) - 'a']++;
            count[s.charAt(i) - 'a']--;
        }

        if (isZero(count)) {
            result.add(0);
        }

        for (int i = m; i < n; i++) {
            count[s.charAt(i) - 'a']--;
            count[s.charAt(i - m) - 'a']++;

            if (isZero(count)) {
                result.add(i - m + 1);
            }
        }

        return result;
    }

    private boolean isZero(int[] count) {
        for (int x : count) {
            if (x != 0) {
                return false;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        String s = sc.next();
        String p = sc.next();

        Solution obj = new Solution();
        System.out.println(obj.findAnagrams(s, p));

        sc.close();
    }
}
